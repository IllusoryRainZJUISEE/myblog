递归

1. 递归(recursion)
(1) 用递归函数计算两个整数的最大公约数
int gcd(int a, int b)
{
   int r;
   r = a % b;
   if(r == 0)
      return b;
   else
      return gcd(b, r);
  
}
main()
{
   int y;
   y = gcd(9, 15);
}

(2) 用递归函数把一个正整数转化成二进制格式输出
解法1:
void f(unsigned int x)
{
   if(x < 2)
      printf("%d", x);
   else
   {
      f(x/2); // 等价于f(x>>1); 
      f(x%2); // 等价于f(x&1);
   }
}
main()
{
   f(6);
}

解法2:
int f(unsigned int x, char *s)
{  int n;
   if(x < 2)
   {
      *s = x + '0';
      return 1;
   }
   else
   {
      n = f(x/2, s);
      n += f(x%2, s+n);
      return n;
   }
}

main()
{
   char a[100];
   int n;
   n = f(6, a); // n表示计算出来的二进制位数, 
                // a用来保存二进制字符串
   a[n] = '\0';
   puts(a);
   
}

/* 显示迷宫 */
void show_maze(int x0, int y0, int x1, int y1)
{
   int x, y;
   unsigned char c;
   for(y=y0; y<=y1; y++)
   {
      for(x=x0; x<=x1; x++)
      {
         c = maze[y][x];
         if(c == 0)
            c = ' ';
         else if(c == 1)
            c = '#';
         putchar(c);
      }
      putchar('\n');
   }
}


/* 画路线 */
void show_trace(void)
{
   int x, y;
   char c;
   for(y=0; y<=BOTTOM_BOUND; y++)
   {
      for(x=0; x<=RIGHT_BOUND; x++)
      {
         if(trace[y][x] == 1)
            c = '.';
         else
         {
            c = maze[y][x];
            if(c == 0)
               c = ' ';
            else if(c == 1)
               c = '#';            
         }
         putchar(c);
      }
      putchar('\n');
   }
}


/* 用深度优先算法搜索迷宫查找目标, (x0,y0)是起点, (x1,y1)是终点 */
int move_bug(int x0, int y0, int x1, int y1)
{
   int k, nx, ny;
   static int d[4][2]={{-1,0},{0,1},{1,0},{0,-1}};
   if(x0<0 || y0<0 || x0>RIGHT_BOUND || y0>BOTTOM_BOUND)
      return 0;
   if(maze[y0][x0] == 1)
      return 0;
   if(mark[y0][x0] == 1)
      return 0;
   mark[y0][x0] = 1;  /* Mark it so that we will not come here again */
                      /* 做个记号, 表示已来过此处, 下次递归时再遇到此处应该回头 */

   trace[y0][x0] = 1; /* This may be a node on the way to the target */
                      /* 暂时把(x0,y0)存放到路线中 */
   if(x0 == x1 && y0 == y1)
   {
      return 1;        /* 到达目标, 返回真 */
   }
   for(k=0; k<4; k++)  /* 尝试让虫子爬到某个方向 */
   {
      nx = x0+d[k][0];
      ny = y0+d[k][1];
      if(move_bug(nx, ny, x1, y1) == 1)
         return 1;       /* 若让虫子从(x0,y0)出发, 顺着第k个方向爬到(nx,ny)并
                            最终能遇到目标则返回真 */
   }
   trace[y0][x0] = 0; /* After 4 loops above, we have proved that this is not 
                         a node on the way to the target */
                      /* 由于4个方向都是错误的, 故在路线中删除此坐标 */
   return 0;
}


main()
{
   show_maze(0, 0, RIGHT_BOUND, BOTTOM_BOUND); /* 显示迷宫 */
   move_bug(1, 1, tx, ty); /* 让虫子在迷宫中用深度优先算法寻找右下角的目标 */
   show_trace();        /* 显示正确路线 */
}


2. 变量的存储类别
(1) 局部动态变量(local automatic variable)
定义在函数体内或函数头中并且不用static修饰的变量称为
局部动态变量，局部动态变量的修饰词是auto(可省略不写)。
局部是指变量的空间特性，动态是指变量的时间特性。
变量的作用域(scope)是指能访问该变量的代码块，
变量的生命期(life time/duration)是指变量从诞生到死亡的时间周期。
局部变量的作用域及生命期均为它所属的代码块。
局部动态变量在定义时若没有赋值，则它的默认值是一个随机数。

int f(int x)
{  y = 0;
   x++;
   return x; // return返回时，会把x的值先保存到CPU内部的一个
             // 全局变量中再带回给main
}
main()
{
   int a=3;
   {  
      int y;
      y = f(a);
      printf("y=%d\n", y);
   }
   printf("y=%d\n", y); // 语法错误
   printf("a=%d\n", a);
}

(2) 全局变量(global variable)
定义在所有函数之外的变量称为全局变量
全局变量的作用域是整个程序，其生命期是从main()开始运行
到main()结束运行。全局变量的定义时若没有赋值，则它的
初始值=0
#include <stdio.h>
int g=100; // int g;等价于int g=0;
void f(void)
{
   g++; // 外面的g=101
}
main()
{  int *p = &g;
   int g=3; // 内部变量优先
   g*=2; // 内部的g=6
   f();
   printf("g=%d", g); //g=6
}

(3) 局部静态变量(local static variable)
定义在函数体内并且用static修饰的变量称为局部静态变量
局部静态变量的作用域是它所属的代码块; 局部静态变量的
生命期是从main()开始运行起到main()结束运行止;
局部静态变量的定义并赋值语句是在main()开始运行时就执行了,
当f()被调用时，该变量的定义语句从不被执行;
当f()调用结束时，该变量并不会死亡; 局部静态变量在定义时
若没有赋值则它的默认值=0。局部动态变量在定义时若没有赋值，
则它的初始值是一个随机数。
例1:
int f(void)                int g(void)        
{                          {                  
   static int a=3;            int a=3; //等价于auto int a=3;
   a++;                       a++;            
   return a;                  return a;       
}                          }                  
main()
{
   int x, y;
   x = f();
   x = f();
   y = g();
   y = g();
}

例2:
char *week(int i)
{
   static char w[7][4] = 
   {
      "Sun", "Mon", "Tue", "Wed", "Thu", "Fri", "Sat"
   };
   return w[i]; // 等价于return &w[i][0];
}
main()
{
   char *p;
   p = week(0);
   puts(p);
}

(4) 全局静态变量
定义在任何函数外面并且用static修饰的变量称为全局静态变量
全局静态变量的作用域是它所属的.c文件，它的生命期是从main()
开始运行起到main()结束运行止。
/* f.c */
static int s=123; // 这个全局静态变量s只能在f.c中引用，不能在main.c中引用
int g=100;        // 这个全局变量g既能在f.c中引用，又能在main.c中引用
void f1(void)
{
   g++; // g=201
   s++; // s=124
}

int f2(void)
{
   g+=2; // g=203
   s+=2; // s=126
   return s;
}

/* main.c */
#include <stdio.h>
extern int g;  // 声明有一个来自于外部的全局变量名叫g
void f1(void); // 声明函数f1()的原型
int  f2(void); // 声明函数f2()的原型
main()
{
   int s_in_f;
   g*=2;
   f1();
   s_in_f = f2(); // main无法直接引用f.c中的s，但可以通过调用f2()来间接获得s的值并保存到s_in_f中
   printf("g=%d, s_in_f=%d\n", g, s_in_f); // g=203, s_in_f=126
}

DevC++中多个.c文件构建项目并编译调试的步骤:
①在桌面上新建一个文件夹并命名为project，并在该文件夹内新建main.c及f.c
②双击Dev-C++图标，选择菜单"文件->新建->项目->Empty Project"
 "项目文件夹" 选择桌面上的 "project"
 "项目名称" 填 "prj"
③点击Dev-C++窗口左侧的"项目管理"，选中"未命名"->右键->移除文件 
④选中prj->右键->添加->选择桌面project文件夹内的main.c及f.c，按住Ctrl键可以同时选择2个文件
⑤点菜单"运行->编译"
⑥点菜单"运行->调试"
