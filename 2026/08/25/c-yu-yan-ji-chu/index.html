4. switch...case
#include <stdio.h>
main()
{
   int a, b, c, y, flag=0;
   scanf("%d %d %d", &a, &b, &c);
   switch(c)
   {
   case 10: // ①case后面的表达式不能包含变量，只能是常数
      y = a + b;
      break; // 这里的break表示当前case的结束
   case 20:  // ②只能判断相等，不能判断< > !=等范围
      y = a - b;
      break; // ③每个case后面必须有一个break
   case 30:
      y = a * b;
      break;
   default:  // 其它case
      flag = 1; // 最后case的break可以省略           
   }
   if(flag == 0)
      printf("y=%d\n", y);
   else
      printf("bad choice");
}

3. ++ --
a=a+1; 可以简化为a++;或++a;
a=a-1; 可以简化为a--;或--a;
当++或--作用的对象又参与了别的运算时，++、--位于左右是有区别的
int a=3, b=5, c;
c = a++ + b; // a++的整体的值等于a的原值
             // c = 3 + 5 = 8; 再计算a=a+1=4
c = ++a + b; // ++a的整体的值等于a+1的值
             // c = 4+5 = 9; 再计算a=a+1=4

int a=3, b;
b = -a--;    // b=-3, a=3-1=2

4. 整数与小数的混合运算
int a=3;
float b=4.95;
float c;
c = a + b;
  (float)a + b // 按整数服从小数规则统一类型
   3.0 + 4.95 = 7.95
c = a; // c = 3.0
a = b; // a=4  小数后面部分全部丢弃即截断

整数的两种类型

1. 整数的两种类型
(1) 符号数(signed number): 包括正整数、负整数、零
计算机中用补码(two`s complement)表示一个符号数
8位正整数的二进制补码的最高位恒为0，其余各位表示该数的值，例如: +127 = 0111 1111
如何用8位二进制补码表示-1？
+1 = 0000 0001           +2 = 0000 0010
-1 = 1111 1111 +)        -2 = 1111 1110
------------------
     0000 0000

+127 = 0111 1111        -128 = 1000 0000
-127 = 1000 0001        +127 = 0111 1111
                        +1   = 0000 0001 +)
                        -------------------
                                       0

8位正整数的取值范围是[0000 0000, 0111 1111]即[0x00,0x7F]或[0, 127]
8位负整数的取值范围是[1000 0000, 1111 1111]即[0x80,0xFF]或[-128, -1]

① 8位符号数即char类型的取值范围是[1000 0000, 0111 1111]
即[0x80, 0x7F]或[-128, +127]
char的输入/输出格式: %c、%hhd、%hhx、%hho
用getchar()这个函数可以对char或unsigned char类型的变量做输入
用putchar()则可以用符号的形式输出某个char或unsigned char

② 16位符号数short int的取值范围[1后面15个0, 0后面15个1]即
[0x8000, 0x7FFF]或[-32768, +32767], 输入/输出格式: 
%hd、%hx、%hX、%ho

③ 32位符号数long int的取值范围[1后面31个0, 0后面31个1]即
[0x80000000, 0x7FFFFFFF]或[-2147483648, +2147483647], 
输入/输出格式: %ld、%lx、%lX、%lo

④ int是对short int、long int的抽象, 它在Dev-C++中等价于
long int，在TC中等价于short int，输入/输出格式: 
%d、%x、%X、%o


-1 的8位二进制补码  =1111 1111
-1 的16位二进制补码 =16个1
-1 的32位二进制补码 =32个1

   char a=0xFE; // 1111 1110

(2) 非符号数(unsigned number): 仅包含零及正整数
① unsigned char 是8位的非符号整数类型
取值范围是[0000 0000, 1111 1111], 即[0x00, 0xFF]或
[0, 255], 输入/输出格式: %c、%hhu、%hhx、%hhX、%hho
其中u表示十进制非符号整数格式, 输出结果一定是非零值
其中%c格式表示是以字符(character)的形式来做输入或输出
② unsigned short int 是16位的非符号整数类型
取值范围是[16个0, 16个1], 即[0x0000, 0xFFFF]或
[0, 65535], 输入/输出格式: %hu、%hx、%hX、%ho, 其中h代表half
③ unsigned long int 是32位的非符号整数类型
取值范围是[32个0, 32个1], 即[0x00000000, 0xFFFFFFFF]或
[0, 4294967295], , 输入/输出格式: %lu、%lx、%lX、%lo
其中o表示八进制(octal)
④ unsigned int 是对unsigned short int及unsigned long int的抽象,
它在某些编译器中如dev-C++中等价于unsigned long int，而在另外
一些编译器中如TC中等价于unsigned short int，输入/输出格式为:
%u、%x、%o

常数混合运算

1. 各种类型的常数表示方式
(1) unsigned char、char类型的常数:
① 'A'  'B'  'C'  '0'  '1'  '2'  '@'  '#'  ' '
② '\n'  '\t'  '\a'  '\b'  '\\'  '\''  '\"'
  printf("%c", '\a'); 或 putchar('\a'); 并不会在屏幕上
  输出任何内容，但会让喇叭发声(a:alert)
  '\b'表示退格(back space)
  printf("ABC\bxyz"); 会输出ABxyz
  '\\' 表示一个反斜杠自己，不能写成'\'
  '\'' 表示一个单引号自己, 不能写成'''
  双引号既可以写成'"'也可以写成'\"'
  printf("%hhu %hhu", '\n', '\t'); // 输出10 9
  printf("\\n"); 输出\n  
③ '\x41'  '\xA'  '\xFF'  十六进制格式
'\x41'表示ASCII码等于十六进制0x41的那个字符即'A'
'\xA' 等价于 '\n'
'\xFF'表示ASCII码等于十六进制0xFF的那个字符即ASCII码表中的最后那个字符

④ '\101' '\12' '\377' '\0' 八进制格式
'\101'表示ASCII等于八进制0101的那个字符即'A'
'\12'表示ASCII等于八进制012的那个字符即'\n'
'\377'表示ASCII等于八进制0377的那个字符即'\xFF'
'\0'表示ASCII等于八进制0的那个字符即ASCII码表中的首个字符

  
  char x = 'A';  // 优先选择
  char x = '\x41';
  char x = '\101';
  char x = 65;
  
  char x = '\n'; // 优先选择
  char x = '\x0A';
  char x = '\12';
  char x = 10;
  
  char x = '\0'; // 优先选择
  char x = '\x0';
  char x = 0;
  // 0是一个int类型的常数，在dev-c++中
  // 它的宽度是32位，赋值时会发生截断
  // a会被赋值为0的低8位, 而0的高24位
  // 则会被丢弃
  0000 0000  0000 0000  0000 0000  0000 0000
  ~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~~  =========
     高24位会被丢弃                低8位赋值给x
  
  printf("\3\3\3"); 或 printf("\x3\x3\x3"); 或 printf("%c%c%c", 3, 3, 3);
  不能改写成printf("333");

(2) int、unsigned int类型的常数
① int a=-123; unsigned int b=456;
② int a=0x41; unsigned int b=0xFF;  // 十六进制(hexadecimal)
③ int a=0101; unsigned int b=0377;  // 八进制(octal)

(3) long int、unsigned int、unsigned long int类型的常数
① long int a = 1L; // (long int)1
  long int a = 1; // 在TC中, 1作为int类型的常数仅有16位
                  // 故在赋值时需要做扩充(左边补16个0)                   
② unsigned int b = 0U; // (unsigned int)0
③ unsigned long int c = 123UL; // (unsigned long int)123

(4) float、double类型的常数
① float a=3.14; double b=3.14;
② float a=3.14E2; double b=3.14E-2; // 科学计数法
③ float a=3.14F;  // 相当于(float)3.14
小数常数的默认类型为double，并非float
如3.14是double类型，要强调3.14是一个float时需要写成3.14F

3. 截断与扩充
(1) 整数的截断: 当把宽度大的值赋值给宽度小的变量时会发生截断，截断的
规则是保留低位，丢弃高位。
short int a=0x1234; // 0x1234=0001 0010  0011 0100
char b;
b = a; // b=0x34
(2) 小数的截断: 当把小数赋值给整型变量时会发生截断, 截断时
仅保留整数部分，丢弃小数部分;
float a=3.95;
int b, c;
b = a;  // b = 3
c = a + 0.5; // c=4 四舍五入

(3) 整数的扩充
把宽度小的整数值赋值给宽度大的整型变量时需要先扩充再赋值，
扩充的规则是正整数左边补0，负整数左边补1。
char a = 127; // a=127=0111 1111
short int b;
b = a;  // b=127=0000 0000 01111 1111
        //                 ~~~~~~~~~~
char a = -2;  // a=-2=1111 1110
short int b;  // b=1111 1111 1111 1110 = -2
b = a;        //             ~~~~~~~~~

unsigned char a=254; // a=254=1111 1110
short int b;  // b=0000 0000 1111 1110 = 254
b = a;        //             ~~~~~~~~~

(4) 当对负整数右移时须在左边补1，对正整数右移时则在左边补0
char a = -4;  //原 a=-4=1111 1100
a = a >> 1;   //新 a=1111 1110=-2


4. 运算符: sizeof  = += -+ *= /= %= <<=  >>=  &= |= ^= ,
sizeof用来求某个类型的变量的宽度是多少字节
int n;
n = sizeof(char); // n = 1
n = sizeof(short int); // n = 2
n = sizeof(long int); // n = 4
n = sizeof(float); // n = 4
n = sizeof(double); // n = 8

int a=2, b, c;
c = b = a; // 相当于c = (b = a);

int a=2, b=3;
b += a;  // 相当于 b = b + a;

int a=3;
a <<= 1; // 相当于 a = a << 1;

逗号运算符
int a=2, b=3, c;
c = (a+=2, b-=1, b*=a); // c=8
    a=4    b=2   b=8

?:
int a=5, b;
b = a > 0 ? 1 : a < 0 ? -1 : 0; 等价于 b = a > 0 ? 1 : (a < 0 ? -1 : 0);
