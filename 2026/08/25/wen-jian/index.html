1. 文件
(1) 例1:读文件内容输出到屏幕上(bug版)
假设abc.txt的内容为"ABC"，则以下程序会在屏幕上输出"ABC\xFF"
#include <stdio.h>
#include <stdlib.h>
int main()
{
   FILE *fp; // fp是文件结构指针变量
   char c;   //文件名    打开方式, "r"表示只读
   fp = fopen("abc.txt", "r"); // fopen()用来打开文件
   if(fp == NULL)
      exit(0); // 终止main()的运行, 0是main()的返回值
               // 此处可以用return 0;代替
   while(!feof(fp)) // feof()用来判断文件内容有没有读光
   {                // 若读光则返回非零(真)，否则返回0(假)
                    // eof: end of file
      c = fgetc(fp);// fgetc()从文件中读取一个char
      putchar(c);
   }
   fclose(fp);   
}
/* 例1中的feof()有滞后，导致多一次循环，从而把EOF当作一个有效字符输出 */

(2) 例2:读文件内容输出到屏幕上(修正版)
假设abc.txt的内容为"ABC"，则以下程序会在屏幕上输出"ABC"
#include <stdio.h>
#include <stdlib.h>
int main()
{
   FILE *fp; // fp是文件结构指针变量
   char c;   //文件名    打开方式, "r"表示只读
   fp = fopen("abc.txt", "r"); // fopen()用来打开文件
   if(fp == NULL)
      exit(0); // 终止main()的运行, 0的main()的返回值
               // 此处可以用return 0;代替
   while((c = fgetc(fp)) != EOF) // feof()用来判断文件内容有没有读光
   {                // 若读光则返回非零(真)，否则返回0(假)
                    // eof: end of file
                    // EOF就是-1，定义在stdio.h中
      putchar(c);
   }
   fclose(fp);   
}

(3) 例3: 用fgetc()、fputc()把文件abc.txt复制到xyz.txt
/*
设abc.txt的内容为:
1AAA
2BBB
3CCC

则xyz.txt会拥有相同内容
*/
#include <stdio.h>
#include <stdlib.h>
int main()
{
   FILE *fp1, *fp2; // fp1、fp2是文件结构指针变量
   char c;   //文件名    打开方式, "r"表示只读
   fp1 = fopen("abc.txt", "r"); // fopen()用来打开文件
   fp2 = fopen("xyz.txt", "w"); // "w"表示只写
   if(fp1 == NULL || fp2 == NULL)
      exit(0); // 终止main()的运行, 0的main()的返回值
               // 此处可以用return 0;代替
   while((c = fgetc(fp1)) != EOF) // feof()用来判断文件内容有没有读光
   {                // 若读光则返回非零(真)，否则返回0(假)
                    // eof: end of file
                    // EOF就是-1，定义在stdio.h中
      fputc(c, fp2); // 把c写入到fp2对应的文件中
      putchar(c);
   }
   fclose(fp1);
   fclose(fp2);
}


(4) 例4: 用fgets()、fputs()把文件abc.txt复制到xyz.txt
/*
设abc.txt的内容为:
1AAA
2BBB
3CCC

则xyz.txt会拥有相同内容
*/
#include <stdio.h>
#include <stdlib.h>
int main()
{
   FILE *fp1, *fp2; // fp1、fp2是文件结构指针变量
   char s[100];   //文件名    打开方式, "r"表示只读
   fp1 = fopen("abc.txt", "r"); // fopen()用来打开文件
   fp2 = fopen("xyz.txt", "w"); // "w"表示只写
   if(fp1 == NULL || fp2 == NULL)
      exit(0); // 终止main()的运行, 0的main()的返回值
               // 此处可以用return 0;代替
   while(!feof(fp1)) // feof()用来判断文件内容有没有读光
   {                // 若读光则返回非零(真)，否则返回0(假)
                    // eof: end of file
                    // EOF就是-1，定义在stdio.h中
      if(fgets(s, sizeof(s)-1, fp1) != NULL) // 从文件fp1中读取长度最多为
      {
                    // 99个char的一行内容,
                    // 保存到数组s中
         fputs(s, fp2);  // 把s中的字符串写到fp2中                         
         printf(s);      // 只保存'\0'左边的内容, 末尾不会添加回车
      }
   }
   fclose(fp1);
   fclose(fp2);
}


(5) 例5: 用fscanf()、fprintf()统计abc.txt中的学生成绩，输出总分及平均分
/*
设abc.txt的内容为:
AAA 90
BBB 80
CCC 70

则xyz.txt会拥有以下内容:
sum=240, average=80

*/
#include <stdio.h>
#include <stdlib.h>
int main()
{
   FILE *fp1, *fp2; // fp1、fp2是文件结构指针变量
   char name[10];
   int score, sum=0, n=0;
   fp1 = fopen("abc.txt", "r"); // fopen()用来打开文件
   fp2 = fopen("xyz.txt", "w"); // "w"表示只写
   if(fp1 == NULL || fp2 == NULL)
      exit(0); // 终止main()的运行, 0的main()的返回值
               // 此处可以用return 0;代替
   while(!feof(fp1)) // feof()用来判断文件内容有没有读光
   {                // 若读光则返回非零(真)，否则返回0(假)
      fscanf(fp1, "%s %d ", name, &score);
      sum += score;
      n++;                          
   }
   fprintf(fp2, "sum=%d, average=%d\n", sum, sum/n);
   fclose(fp1);
   fclose(fp2);
}

(7) 文件的打开方式: "r" "w" 
"a": append, 追加方式, 是特殊的写入方式
   原文件可以存在也可以不存在
   若已存在，并不会删除原文件的内容
   文件内部指针会移到EOF处，新写入内容会添加到原文件末尾
   设原文件内容为:
   ABC
   用fputc('X', fp);
   则新文件内容变成:
   ABCX

"r+":读写方式，要求原文件已存在，不会删除原文件内容
"w+":读写方式，会删除原文件内容
"a+":读写方式，不会删除原文件内容
"rb":二进制只读方式
"wb":二进制只写方式
"ab":二进制追加方式
"rb+":二进制读写方式
"wb+":二进制读写方式
"ab+":二进制读写方式

不含"b"的打开方式即文本方式会在读文件时把0x0D与0x0A的整合成0x0A,
而在写文件时则会把0x0A化成0x0D、0x0A写入;
含有"b"的打开方式称为二进制方式, 无论读写文件都不会做上述转化

(8) 用fread()、fwrite()把文件abc.txt复制到xyz.txt
/*
设abc.txt的内容为:
AAA 90
BBB 80
CCC 70

则xyz.txt会拥有一样内容
*/
#include <stdio.h>
#include <stdlib.h>
main()
{
   FILE *fp1,*fp2;
   long len;
   char *p;
   fp1 = fopen("abc.txt", "rb");
   fp2 = fopen("xyz.txt", "wb");
   fseek(fp1, 0, SEEK_END); // 把文件内部指针移到EOF处
   len = ftell(fp1); // 返回内部指针与首字节的距离即文件长度
   fseek(fp1, 0, SEEK_SET); // 相当于rewind(fp1);
   p = malloc(len);
   fread(p, 1, len, fp1); // 读取fp1的内容到p指向的内存块中
   fwrite(p, 1, len, fp2); // 把p指向的内容写入fp2文件
   fclose(fp1);
   fclose(fp2);
}

2. 计算1+2*3/4的值
(1) 解法1
#include <stdio.h>
int main()
{
   double sum=0;
   int x;
   char b='+', c;
   do
   {
      scanf("%d%c", &x, &c); 
      if(b == '+')
         sum += x;
      else if(b ==  '-')
         sum -= x;
      else if(b == '*')
         sum *= x;
      else if(b == '/')
         sum /= x;
      b = c;
   } while(c != '\n');
   printf("sum=%lf\n", sum);
}

(2) 解法2(用函数指针)
#include <stdio.h>

double add(int a, int b)
{
   return a+b;
}

double sub(int a, int b)
{
   return a-b;
}

double mul(int a, int b)
{
   return a*b;
}

double div(int a, int b)
{
   return (double)a/b;
}

typedef double FUN(int a, int b);

int g(char c)
{
   char t[]="+-*/";
   int i;
   for(i=0; i<4; i++)
   {
      if(t[i] == c)
         return i;
   }
   return 0;
}

int main()
{
   double sum=0;
   FUN *p[4]={add, sub, mul, div}; // p是一个数组, 每个元素均为函数指针
                                   // 此处的add、sub、mul、div作为函数名均表示该函数的地址
   
   int x, i;
   char b='+', c;
   do
   {
      scanf("%d%c", &x, &c); 
      i = g(b); // 把+ - * /转化成0 1 2 3
      sum = (*p[i])(sum, x); // 当i=0时, (*p[i])就是函数add本身
      // 以上两行还可以进一步简化为: sum = (*p[g(b)])(sum, x);
      b = c;
   } while(c != '\n');
   printf("sum=%lf\n", sum);
}
