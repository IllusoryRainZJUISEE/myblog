### 结构
```c
1. 结构(structure)类型、结构变量、结构数组
struct st // struct st是结构类型名, st是标签名
{
   int num;   // num与score均为结构成员(member)
   int score;
};

main()
{   
   struct st a, b, c[4]; // a、b是结构类型的变量, c是结构数组
   int i;
   for(i=0; i<4; i++)
   {
      scanf("%d %d", &c[i].num, &c[i].score);
   }
   for(i=0; i<4; i++)
   {
      printf("%d %d", c[i].num, c[i].score);
   }
   a.num = 1;
   a.score = 90;
   b = a;
   //以上这条语句相当于以下2句话:
   b.num = a.num;
   b.score = a.score;   
}

2. 结构指针
struct st
{
   char name[10];       
   int score;
};
main()
{
   struct st a={"Tom", 90}, b={"Jerry", 80}, c[4];
   struct st *p; // p是结构指针变量
   int i; 
   p = &c[0]; // 或写成p = c;
   for(i=0; i<4; i++)
   {                 // &c[i].name[0]与c[i].name等价
      //① scanf("%s %d", c[i].name, &c[i].score);
                     // &c[i].name 是不合适的写法, 因为&c[i].name的类型为char (*)[10]并非char *
                     
      //② scanf("%s %d", (*(p+i)).name, &(*(p+i)).score); // ②与①等效
      
      scanf("%s %d", (*p).name, &(*p).score); // ③
      // 或
      scanf("%s %d", p->name, &p->score); // ④和③等价
      
      // p-> ≡ (*p).   表示p所指向的对象的
      // .的左边必须是结构变量，->的左边必须是结构指针
      // 它们的共同点是: 右边均为结构成员
      p++;
   }   
}

   char *t = "xyz"; // 正确, t是指针变量, t被赋值为&"xyz"[0]
   s = "ABC"; // 语法错误, 因为这里的s≡&s[0]是个常数
   strcpy(s, "ABC"); // 正确
   s[0]='A'; s[1]='B'; s[2]='C'; s[3]='\0'; //正确
   
   AST a;
   PST b;

   scanf("%s %d", a.name, &a.score); // 正确
   

   scanf("%s %d", b.name, &b.score); // 错误，因为b.name是野指针

#include <stdlib.h>
main()
{
   int n=10;

   char *a;

   int i;
   a = malloc(n); // malloc()会分配长度为n字节的内存块

   for(i=0; i<n; i++)

6. 链表(list)
#include <stdio.h>
#include <stdlib.h>
struct st
{
   int num, score;
};

main()
{
   struct st *p, *q, *r;
   p = malloc(sizeof(struct st));
   p->num = 1;
   p->score = 90;
   
   q = malloc(sizeof(struct st));
   q->num = 2;
   q->score = 80;
   
   r = malloc(sizeof(struct st));
   r->num = 3;
   r->score = 70;
   
   p->next = q;
   q->next = r;
   r->next = NULL;
   
   while(p != NULL)
   {
      printf("num=%d, score=%d\n", p->num, p->score);
      p = p->next;
   }
}

7. 创建一个链表、在链表中插入一个节点、输出链表中的各个节点
/* 图解: http://cc.zju.edu.cn/bhh/listadd.png */
#include <stdio.h>
#include <stdlib.h>
struct st 
{
   int num;                  /* 学号 */
   int score;                /* 成绩 */
   struct st *next;          /* next是指向下个节点的指针 */
};

struct st *create(void);
void show(struct st *p);
struct st *insert(struct st *p, struct st *pnew);

main()
{
   struct st *head, *pnew;
   puts("*********************");
   head = create();          /* 建立链表，返回表首地址(即首节点的地址) */
   puts("*********************");
   pnew = malloc(sizeof(struct st));
                             /* pnew指向新节点 */
   scanf("%d %d", &pnew->num, &pnew->score);
                             /* 输入新节点的成员信息 */
   pnew->next = NULL;        /* 新节点后面无节点，故赋值为NULL */
   head = insert(head, pnew);/* 插入节点 */
   puts("*********************");
   show(head);               /* 输出链表 */
}

struct st *insert(struct st *p, struct st *pnew)
{
   struct st *phead, *pfore;
   phead = p;                /* 保存首节点的地址到phead */
   pfore = NULL;             /* 首节点前面无节点，故pfore赋值为NULL */
   while(p != NULL) 
   {
      if(pnew->num <= p->num)/* 若新节点的学号 ≤ 当前节点的学号... */
         break;              /* 则跳出 */
      pfore = p;             /* 把当前节点的地址赋值给pfore，使得下次循环时pfore刚好指向前一节点 */
      p = p->next;           /* 使p指向下一个节点 */
   }
   if(pfore == NULL)         /* 若插入点位于首节点处 */
   {      
      pnew->next = p;        /* 使新节点的next指向原首节点，也可写成pnew->next = phead; */
      phead = pnew;          /* 更新phead，使phead指向新节点 */
   }
   else                      /* 若插入点不在首节点处，即新节点为中间节点或尾节点 */
   {
      pnew->next = pfore->next;        
                             /* 使新节点的next指向插入点处的节点, 也可以写成pnew->next = p; */
      pfore->next = pnew;    /* 使插入点前面那个节点的next指向新节点 */
   }
   return phead;             /* 返回表首地址 */
}

struct st *create(void)
{
   struct st *phead, *pfore, *pnew;
   phead = pfore = pnew = NULL;
   do 
   {
      pnew = malloc(sizeof(struct st));
                             /* 分配一块内存用来存放一个节点，pnew=内存块的地址 */
      scanf("%d %d", &pnew->num, &pnew->score);
                             /* 输入学号与成绩存放到刚分配的节点中 */
      pnew->next = NULL;     /* 新节点的下一个节点暂时不存在，故赋值为NULL */
      if(pnew->num == 0)     /* 若学号为0表示当前结节作废且后续不再有输入 */ 
      { 
         free(pnew);         /* 释放新节点占用的内存 */
         pnew = NULL;        /* pnew作废 */
      }
      if(phead == NULL)      /* 若表首指针为空，则新节点为首节点 */
         phead = pnew;      
      else
         pfore->next = pnew; /* 把新节点与前一节点连接起来 */      
      pfore = pnew;          /* 保存新节点地址到pfore，使得下次循环时现在的pfore刚好指向前一节点 */
   } while (pnew != NULL);   /* 若新节点未作废则继续循环 */
   return phead;             /* 返回表首地址 */
}

void show(struct st *p)
{
   while(p != NULL) 
   {
      printf("%d %d\n", p->num, p->score);
      p = p->next;           /* 使p指向下一个节点 */
   }
}

8. 删除链表中的一个节点
/* 图解: http://cc.zju.edu.cn/bhh/listdel.png */
#include <stdio.h>
#include <stdlib.h>
struct st 
{
   int num;                  /* 学号 */
   int score;                /* 成绩 */
   struct st *next;          /* next是指向下个节点的指针 */
};
struct st *create(void);
void show(struct st *p);
struct st *delete(struct st *p, int num);

main()
{
   struct st *head;
   int num;
   puts("*********************");
   head = create();          /* 建立链表，返回表首地址(即首节点的地址) */
   puts("*********************");
   scanf("%d", &num);
   head = delete(head, num); /* 删除节点 */
   puts("*********************");
   show(head);               /* 输出链表 */
}

struct st *delete(struct st *p, int num)
{
   struct st *phead, *pfore;
   phead = p;                /* 保存首节点的地址到phead */
   pfore = NULL;             /* 首节点前面无节点，故pfore赋值为NULL */
   while(p != NULL) 
   {
      if(p->num == num)      /* 若找到学号等于num的节点... */
         break;              /* 就跳出 */
      pfore = p;             /* 保存当前节点的地址到pfore，使得下次循环时现在的pfore刚好指向前一节点 */
      p = p->next;           /* 使p指向下一个节点 */
   }
   if(p == NULL)             /* 若没找到学号等于num的节点... */
      return phead;          /* 就直接返回首节点的地址 */
   if(pfore == NULL)         /* if(pfore == NULL)等价于if(p == phead)，含义均为：若找到的节点是首节点 */
   {                                                
      phead = p->next;       /* 使phead指向原首节点后面那个节点 */
      free(p);               /* 释放首节点占用的内存 */
   }
   else                      /* 若找到的节点不是首节点 */
   {
      pfore->next = p->next; /* 使前一节点指向找到节点的后一节点 */      
      free(p);               /* 释放找到节点占用的内存 */
   }
   return phead;             /* 返回表首地址 */
}

struct st *create(void)
{
   struct st *phead, *pfore, *pnew;
   phead = pfore = pnew = NULL;
   do 
   {
      pnew = malloc(sizeof(struct st));
                             /* 分配一块内存用来存放一个节点，pnew=内存块的地址 */
      scanf("%d %d", &pnew->num, &pnew->score);
                             /* 输入学号与成绩存放到刚分配的节点中 */
      pnew->next = NULL;     /* 新节点的下一个节点暂时不存在，故赋值为NULL */
      if(pnew->num == 0)     /* 若学号为0表示当前结节作废且后续不再有输入 */ 
      { 
         free(pnew);         /* 释放新节点占用的内存 */
         pnew = NULL;        /* pnew作废 */
      }
      if(phead == NULL)      /* 若表首指针为空，则新节点为首节点 */
         phead = pnew;      
      else
         pfore->next = pnew; /* 把新节点与前一节点连接起来 */      
      pfore = pnew;          /* 保存新节点地址到pfore，使得下次循环时现在的pfore刚好指向前一节点 */
   } while (pnew != NULL);   /* 若新节点未作废则继续循环 */
   return phead;             /* 返回表首地址 */
}

void show(struct st *p)
{
   while(p != NULL) 
   {
      printf("%d %d\n", p->num, p->score);
      p = p->next;           /* 使p指向下一个节点 */
   }
}

(1) 例1: 无限递归
#include <stdio.h>
main()

(2) 例2: 求n!
#include <stdio.h>
int f(int n)
{
   if(n==1)
      return 1;
   else
      return n*f(n-1);
}

main()
{
   int y;
   y = f(3);
   printf("y=%d\n", y);
}
```
