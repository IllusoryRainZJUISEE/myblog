   for(i=0; i<n-1; i++)
   {
      for(j=0; j<n-1-i; j++)
      {
         if(a[j] > a[j+1])
         {

         }
      }

}

2. 二分法(折半法)
#include <stdio.h>
main()
{
   int a[5] = {11, 22, 33, 44, 55};
   int x=44, n=5;
   int left, right, mid;
   left = 0;
   right = n-1;
   mid = (left+right)/2;
   while(left<=right && a[mid] != x)
   {
      if(x < a[mid])
         right = mid-1;
      else
         left = mid+1;
      mid = (left+right)/2;   
   }
   if(left <= right)
      printf("%d is found at %d\n", x, mid);
   else
      printf("%d is not found\n", x);
}

4. 已知一个十六进制字符串，把它转化成一个整数
#include <stdio.h>
main()
{
   char a[]="3E8";
   unsigned int x=0, d;
   int i=0;
   while(a[i] != '\0')
   {
      if(a[i] >= '0' && a[i] <= '9')
         d = a[i] - '0';
      else if(a[i] >= 'A' && a[i] <= 'F')
         d = a[i] - 'A' + 10;
      else if(a[i] >= 'a' && a[i] <= 'f')
         d = a[i] - 'a' + 10;
      x = x * 16 + d; // 若n已知, x = x + d*pow(16, n-1-i);
      i++; 
   }
   printf("x=%u\n", x);   
}

5. 数组作为函数的参数
(1) 一维数组作函数的参数
#include <stdio.h>
int f(int a[], int n) // 这里的a是一维数组做形参
{                     // 也可以写成int a[3]、int a[4]
   int i, sum=0;      // 或int a[任意正整数]  
   for(i=0; i<n; i++)
      sum += a[i];
   return sum;
}

main()
{  int d[4] = {1, 2, 3, 4};
   int x[3] = {10, 20, 30}, y;
   y = f(x, 3) + f(d, 4); 
   // x是一维数组做实参, 不能写成x[]或x[0]
}  // 实参数组名可以与形参数组名相同

6. 字符串函数: strlen、strcpy、strcat、strcmp
#include <string.h>
#include <stdio.h>
int my_strlen(char s[])
{
   int i=0;
   while(s[i] != '\0')
      i++;
   return i;
}
main()
{
   char a[10]="ABC";
   int n, m;
   n = strlen(a); // n = 3 = a中'\0'左边的元素个数
   n = my_strlen(a); // n = 3
   m = sizeof(a); // m = 10
}

#include <string.h>
#include <stdio.h>
void my_strcpy(char t[], char s[]) // void表示函数无返回值
{
   int i=-1;
   do
   {
      i++;
      t[i] = s[i];      
   } while (s[i] != '\0');
}
main()
{
   char a[10] = "ABC", b[10] = "012345";
   my_strcpy(b, a);         //  ABC\0
   strcpy(b, a); // 把a的内容复制到b中
   puts(b); // 输出ABC及回车
}


#include <string.h>
#include <stdio.h>
void my_strcat(char t[], char s[]) // void表示函数无返回值
{
   int n = strlen(t);
   int i = -1;
   do
   {
      i++;
      t[n+i] = s[i];
   } while(s[i] != '\0');
}
main()
{
   char a[10] = "ABC", b[10] = "123", c[10]="123";
   my_strcat(c, a);  //            ABC\0
   strcat(b, a); // 把a的内容追加到b的末尾
   puts(b); // 输出123ABC并回车
   puts(c); // 输出123ABC并回车
}

#include <string.h>
#include <stdio.h>
int my_strcmp(char s[], char t[])
{
   int i = 0, d;
   while((d = s[i] - t[i]) == 0 && s[i] != '\0' && t[i] != '\0')
      i++;
   return d;
}
main()
{
   char a[10]="BBC", b[10]="ABC";
   int n;
   n = my_strcmp(a, b);
